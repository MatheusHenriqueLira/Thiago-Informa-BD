CREATE TABLE Turma 
( 
 idturma INT PRIMARY KEY AUTO_INCREMENT,  
 Nome VARCHAR,  
); 

CREATE TABLE Projeto 
( 
 Idpost INT PRIMARY KEY AUTO_INCREMENT,  
 Imagem (x64) VARCHAR,  
 datacriacao DATETIME,  
 Titulo VARCHAR,  
 Conteudo VARCHAR,  
 idAdmin INT,  
); 

CREATE TABLE Informativos 
( 
 Idinfo INT PRIMARY KEY AUTO_INCREMENT,  
 datacriacao DATETIME,  
 Titulo VARCHAR,  
 Conteudo VARCHAR,  
 idAdmin INT,  
 estafixado BOOLEAN,  
 idImagem INT,  
); 

CREATE TABLE Responsavel 
( 
 idresp INT PRIMARY KEY AUTO_INCREMENT,  
 Nome VARCHAR,  
); 

CREATE TABLE ResponsavelTurma 
( 
 idResponsavel INT,  
 idTurma INT,  
); 

CREATE TABLE Admin 
( 
 Id INT PRIMARY KEY AUTO_INCREMENT,  
 Nome VARCHAR,  
); 

CREATE TABLE Imagem 
( 
 x64 INT PRIMARY KEY,  
); 

CREATE TABLE ImgProjeto 
( 
 idImagem INT,  
 idProjeto INT,  
); 

CREATE TABLE ProjetoTurma 
( 
 idProjeto INT,  
 idTurma INT,  
);
